cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
endif()
project(NoLiveArea C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=gnu11 -Wl,-q -Wall -O2 -fno-builtin -fno-tree-loop-distribute-patterns -nostdlib")
add_executable(NoLiveArea nolivearea.c)
target_link_libraries(NoLiveArea taihen_stub SceLibKernel_stub SceIofilemgr_stub SceKernelThreadMgr_stub SceKernelModulemgr_stub ScePower_stub SceAppMgr_stub)
option(NLA_DEBUG "tracing hooks, per-event log and probes (never in a release)" OFF)
if(NLA_DEBUG)
  target_compile_definitions(NoLiveArea PRIVATE NLA_DEBUG=1)
endif()
set_target_properties(NoLiveArea PROPERTIES LINK_FLAGS "-nostdlib")
vita_create_self(NoLiveArea.suprx NoLiveArea CONFIG exports.yml UNSAFE)
